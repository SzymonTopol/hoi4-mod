com_scramble_map_mode = {
    name = "Commonwealth Civil War Scramble"
    
    # We want to color the map at the state level
    color_mode = state 
    
    # The map mode will only be visible to the player if the civil war hasn't fired yet
    is_visible = {
        NOT = { has_global_flag = ready_for_COM_civil_war }
        OR = { tag = CCO tag = CLB tag = CMO tag = CNA tag = CCM }
    }

    # ==========================================
    # CCO (Conservatives) - Highlight Color
    # ==========================================
    state_color = {
        color = { 0 50 255 125 } # Blue (RGBA format: Red, Green, Blue, Transparency)
        trigger = {
            has_state_flag = marked_by_CCO
        }
    }

    # ==========================================
    # CLB (Liberals) - Highlight Color
    # ==========================================
    state_color = {
        color = { 255 200 0 125 } # Yellow
        trigger = {
            has_state_flag = marked_by_CLB
        }
    }

    # ==========================================
    # CMO (Monarchists) - Highlight Color
    # ==========================================
    state_color = {
        color = { 150 0 150 125 } # Purple
        trigger = {
            has_state_flag = marked_by_CMO
        }
    }

    # ==========================================
    # CNA (Fringe) - Highlight Color
    # ==========================================
    state_color = {
        color = { 0 255 0 125 } # Green
        trigger = {
            has_state_flag = marked_by_CNA
        }
    }

    # ==========================================
    # CCM (Communists) - Highlight Color
    # ==========================================
    state_color = {
        color = { 255 0 0 125 } # Red
        trigger = {
            has_state_flag = marked_by_CCM
        }
    }
    
    # ==========================================
    # Default Color (Unclaimed States)
    # ==========================================
    state_color = {
        color = { 100 100 100 50 } # Semi-transparent grey
        trigger = {
            is_owned_by = COM
            NOT = { has_state_flag = com_cw_state_marked }
        }
    }
}
