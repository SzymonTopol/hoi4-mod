add_namespace = commonwealth

# Hidden starter event to accurately trigger the chain 75-100 days from game start
country_event = {
	id = commonwealth.0
	hidden = yes
	
	trigger = {
		tag = COM
		date > 1935.12.31 # Fires as soon as the game starts
	}
	
	fire_only_once = yes
	
	immediate = {
		# 75 days base delay + up to 25 random days = 75 to 100 days
		country_event = { id = commonwealth.1 days = 75 random_days = 25 }
	}
}

country_event = { #monarch death
	id = commonwealth.1
	title = commonwealth.1.t
	desc = commonwealth.1.d
	picture = GFX_report_event_generic_funeral

	is_triggered_only = yes

	immediate = {
		# Swaps the tree immediately upon event pop-up, canceling current focus
		load_focus_tree = empty_focus
	}

	option = {
		name = commonwealth.1.a

		hidden_effect = {
			COM = {
				# 1 day base + up to 6 random days = 1 to 7 days
				country_event = { id = commonwealth.2 days = 1 random_days = 6 }
			}
		}
	}
}
