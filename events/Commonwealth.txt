country_event = {
    id = commonwealth.7
    title = commonwealth.7.t
    desc = commonwealth.7.d
    picture = GFX_report_event_generic_conflict

    is_triggered_only = yes

    immediate = {

        load_focus_tree = empty_focus
        
        88 = { add_core_of = CCO }
        807 = { add_core_of = CLB }
        259 = { add_core_of = CMO }

        CCO = { transfer_state = 88 }
        CLB = { transfer_state = 807 }
        CMO = { transfer_state = 259 }
        
        if = {
            limit = { has_country_flag = COM_chose_conservatives }
            CCO = { change_tag_from = ROOT }
        }
        else_if = {
            limit = { has_country_flag = COM_chose_liberals }
            CLB = { change_tag_from = ROOT }
        }
        else_if = {
            limit = { has_country_flag = COM_chose_monarchists }
            CMO = { change_tag_from = ROOT }
        }

        # 5. Clean up the flags
        clr_country_flag = COM_chose_conservatives
        clr_country_flag = COM_chose_liberals
        clr_country_flag = COM_chose_monarchists
    }
    option = {
         name = commonwealth.7.a
    }
}
